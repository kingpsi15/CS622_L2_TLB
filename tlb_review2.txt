+uint32_t
+TLB::l2SetIndex(Addr vpn) const
+{
+    if (l2Sets == 1)
+        return 0;
+    Addr idxMask = l2Sets - 1;
+    switch (l2HashKind) {
+      case L2HashKind::Mod:
+        return (uint32_t)(vpn % l2Sets);
+      case L2HashKind::Xor7: {
+        uint32_t lo = (uint32_t)(vpn & idxMask);
+        uint32_t hi = (uint32_t)((vpn >> l2SetBits) & idxMask);
+        return lo ^ hi;
+      }
+      case L2HashKind::XorFold: {
+        uint32_t h = 0;
+        Addr v = vpn;
+        while (v) {
+            h ^= (uint32_t)(v & idxMask);
+            v >>= l2SetBits;
+        }
+        return h;
+      }
+    }
+    return 0;
+}

+uint64_t
+TLB::l2Rand()
+{
+    // xorshift64*; local to this TLB instance so we never touch libc's
+    // global rand()/srand() state shared with the rest of the simulator.

+uint32_t
+TLB::l2PlruVictim(uint32_t set) const
+{
+    uint32_t levels = log2u(l2Ways);
+    uint32_t base = set * (l2Ways - 1);
+    uint32_t node = 0, pos = 0;
+    for (uint32_t lvl = 0; lvl < levels; lvl++) {
+        bool bit = l2Plru[base + node] != 0;
+        pos = (pos << 1) | (bit ? 1u : 0u);
+        node = 2 * node + 1 + (bit ? 1u : 0u);
+    }
+    return pos;
+}

+void
+TLB::l2TouchPlru(uint32_t set, uint32_t way)
+{
+    uint32_t levels = log2u(l2Ways);
+    uint32_t base = set * (l2Ways - 1);
+    uint32_t node = 0;
+    for (uint32_t lvl = 0; lvl < levels; lvl++) {
+        uint32_t bit = (way >> (levels - 1 - lvl)) & 1u;
+        // Point the node away from the branch we just used, so the next
+        // victim search walks toward the opposite (least-recently-used)
+        // subtree.
+        l2Plru[base + node] = bit ? 0 : 1;
+        node = 2 * node + 1 + bit;
+    }
